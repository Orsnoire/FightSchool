test('review: retained canonical 18-entry history upgrades to 27 without rewriting its receipts or old results',async()=>{
  const f=await dataCostFixture(2,1,17);
  try {
    const entries=JSON.parse(readFileSync('migrations/cloudflare/meta/_journal.json','utf8')).entries;
    await f.pg.exec('CREATE SCHEMA drizzle; CREATE TABLE drizzle.__drizzle_migrations(id SERIAL PRIMARY KEY,hash text NOT NULL,created_at bigint)');
    for(const e of entries.slice(0,18))await f.pg.query('INSERT INTO drizzle.__drizzle_migrations(hash,created_at) VALUES($1,$2)',[createHash('sha256').update(readFileSync(`migrations/cloudflare/${e.tag}.sql`)).digest('hex'),e.when]);
    const oldLedger=(await f.pg.query('SELECT * FROM drizzle.__drizzle_migrations ORDER BY id')).rows;
    const oldResults=(await f.pg.query('SELECT id,session_id,student_id,xp_earned,gold_reward,totals FROM combat_results ORDER BY id')).rows;
    await migrate(drizzle(f.pg),{migrationsFolder:'migrations/cloudflare'});
    const ledger=(await f.pg.query('SELECT * FROM drizzle.__drizzle_migrations ORDER BY id')).rows;
    assert.equal(ledger.length,27);assert.deepEqual(ledger.slice(0,18),oldLedger);
    assert.deepEqual((await f.pg.query('SELECT id,session_id,student_id,xp_earned,gold_reward,totals FROM combat_results ORDER BY id')).rows,oldResults);
    await migrate(drizzle(f.pg),{migrationsFolder:'migrations/cloudflare'});
    assert.deepEqual((await f.pg.query('SELECT * FROM drizzle.__drizzle_migrations ORDER BY id')).rows,ledger);
  }finally{await f.pg.close();}
});

test('review: unjournaled math templates block cleanup and roll back earlier deletions',async()=>{
  const {f,manifest,query}=await fixture();
  try{
    await query('INSERT INTO math_templates(id,teacher_id,create_key,latest_revision) VALUES($1,$2,$3,1)',[randomUUID(),fight.teacherId,randomUUID()]);
    await assert.rejects(cleanupDatabase(query,manifest,{apply:true}),/foreign key constraint/);
    assert.equal((await query('SELECT count(*)::int n FROM students')).rows[0].n,2);
    assert.equal((await query('SELECT count(*)::int n FROM math_templates')).rows[0].n,1);
  }finally{await f.pg.close();}
});
